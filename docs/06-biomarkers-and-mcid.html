6  Longitudinal Biomarkers and Clinical Meaningfulness (MCID)

  • Show All Code

Statistical significance in biomarker changes does not automatically establish clinical benefit. This chapter covers Minimally Clinically Important Difference (MCID) visualizations, patient response trajectories across clinical severity bands, and real longitudinal proteomics timecourses.

6.1 Minimally Clinically Important Difference: type = "mcid"

The Minimally Clinically Important Difference (MCID) defines the smallest change on a clinical score or patient-reported outcome that patients perceive as beneficial:

\text{Responder} = \mathbb{I}(\Delta \text{Score}_i \ge \text{MCID})

The MCID Dumbbell Plot connects observed clinical effect estimates to validated instrument thresholds (e.g. FACT-G, QLQ-C30, Pain NRS), scaling point glyphs by sample size (N_{\text{treated}}) and sorting endpoints by their effect-to-MCID ratio:

Code 
library(cliomicplot)
library(ggplot2)

mcid_oncology <- data.frame(
  Endpoint = c("FACT-G QoL (Anti-PD1)", "Pain NRS (Analgesic Add-on)",
               "Fatigue Score (Exercise Co-care)", "QLQ-C30 Physical (Anti-PD1)",
               "Appetite Loss (Standard Care)"),
  Estimate = c(8.5, 3.4, 4.2, 11.2, 1.8),
  MCID     = c(5.0, 2.0, 3.0,  8.0, 4.0),
  N        = c(520, 380, 240,  510, 190),
  Category = c("Immunotherapy Exceeding MCID", "Adjuvant Exceeding MCID",
               "Supportive Care Exceeding MCID", "Immunotherapy Exceeding MCID",
               "Standard Care Below MCID")
)

clitheme("nature")

p_mcid <- cliplot(
  Estimate ~ Endpoint,
  data = mcid_oncology,
  type = type_mcid(
    mcid          = "MCID",
    n_treated     = "N",
    category      = "Category",
    sort_by_ratio = TRUE
  ),
  title    = "Minimal Clinically Important Difference (MCID) Evaluation",
  subtitle = "Comparing observed clinical effects to validated instrument thresholds",
  xlab     = "Absolute Effect Estimate (|estimate|)"
)
p_mcid

6.2 Longitudinal Trajectories with Severity Bands: type = "trajectory"

Patient biomarkers exhibit substantial between-subject variability. type_trajectory() overlays semi-transparent individual patient profiles onto shaded reference severity bands (e.g., Normal, Elevated, Critical), anchored by a bold group mean ribbon:

Code 
set.seed(42)
n_patients <- 20
visits <- c("Baseline", "Cycle 2", "Cycle 4", "Cycle 6", "Follow-up")
traj_df <- expand.grid(Patient = paste0("PT-", sprintf("%02d", 1:n_patients)), Visit = visits)

# Simulate biomarker decline under targeted therapy
base_vals <- rep(runif(n_patients, 55, 90), each = length(visits))
decay_vals <- rep(c(0, 15, 32, 48, 55), times = n_patients)
noise <- rnorm(nrow(traj_df), mean = 0, sd = 5)
traj_df$Marker <- pmax(2, pmin(100, base_vals - decay_vals + noise))

clitheme("science")

p_traj <- cliplot(
  Marker ~ Visit | Patient,
  data = traj_df,
  type = type_trajectory(
    severity_bands  = list(Normal = c(0, 25), Elevated = c(25, 50), Critical = c(50, 100)),
    severity_colors = c(Normal = "#E8F8F5", Elevated = "#FEF9E7", Critical = "#FADBD8"),
    show_mean       = TRUE,
    show_ci         = TRUE,
    show_n          = TRUE
  ),
  title    = "Longitudinal Biomarker Response Trajectories",
  subtitle = "Circulating Protein Dynamics with Reference Severity Stratification",
  ylab     = "Biomarker Concentration (ng/mL)"
) +
  theme(plot.margin = margin(5.5, 22, 5.5, 5.5, "pt"))

p_traj

6.3 Time-Course Translational Proteomics (xOmicsShiny Dataset)

Using the bundled StrokeBrain_TimeCourse.RData dataset, we track the temporal kinetics of the macrophage/microglia activation marker Mpeg1 across recovery time points in ischemic stroke (StrkIp), sham surgery (ShamIp), and naive control (StrkC):

Code 
stroke_path <- system.file("extdata", "StrokeBrain_TimeCourse.RData", package = "cliomicplot")
stroke_env <- new.env()
load(stroke_path, envir = stroke_env)

# Merge gene annotations
stroke_long <- merge(
  stroke_env$data_long,
  stroke_env$ProteinGeneName[, c("UniqueID", "Gene.Name")],
  by = "UniqueID",
  all.x = TRUE
)

# Filter for Mpeg1 microglia marker and identify individual mouse replicates
mpeg1 <- subset(stroke_long, Gene.Name == "Mpeg1")
mpeg1$Day <- as.numeric(mpeg1$conc)
mpeg1$Treatment <- mpeg1$treatment
mpeg1 <- mpeg1[order(mpeg1$Treatment, mpeg1$Day, mpeg1$sampleid), ]
mpeg1$Individual <- paste0(mpeg1$Treatment, "_Mouse", ave(seq_len(nrow(mpeg1)), mpeg1$Treatment, mpeg1$Day, FUN = seq_along))

clitheme("science")

# 1 line for each individual mouse replicate
p_tc <- cliplot(
  expr ~ Day | Treatment,
  data = mpeg1,
  group = "Individual",
  type = type_lines(linewidth = 1.1, add_points = TRUE, point_size = 2.4),
  palette = "jco",
  title   = "Mpeg1 Macrophage Marker Time Course",
  subtitle = "Stroke Brain Proteomics — Individual Mouse Replicate Trajectories (Days 1 to 28)",
  xlab    = "Days Post-Onset",
  ylab    = "Protein Expression (LFQ Intensity)"
)
p_tc

When evaluating treatment-level pharmacokinetics without individual profile tracking, type_lines() automatically calculates the group mean trajectory, offsets overlapping error bars horizontally with dodge, highlights mean markers with white borders, and maps individual replicate observations into the background:

Code 
library(scales)

# Mean trajectory with horizontal dodge and standard error bars
p_tc_mean <- cliplot(
  expr ~ Day | Treatment,
  data = mpeg1,
  type = type_lines(
    linewidth        = 1.25,
    show_se          = TRUE,
    dodge            = 0.50,
    errorbar_width   = 0.40,
    add_points       = TRUE,
    point_size       = 2.0,
    point_alpha      = 0.35,
    show_mean_points = TRUE,
    mean_point_size  = 3.6
  ),
  palette = "jco",
  title   = "Mpeg1 Macrophage Marker Mean Kinetics",
  subtitle = "Stroke Brain Proteomics — Mean ± SEM Trajectory Across Recovery (xOmicsShiny)",
  xlab    = "Recovery Timeline Post-Ischemic Stroke",
  ylab    = "Protein Expression (LFQ Intensity)"
) +
  scale_x_continuous(
    breaks = c(1, 3, 7, 14),
    labels = c("Day 1\n(Acute)", "Day 3\n(Subacute)", "Day 7\n(Peak)", "Day 14\n(Resolution)")
  ) +
  scale_y_continuous(
    labels = label_number(scale = 1e-3, suffix = "k"),
    limits = c(0, NA),
    expand = expansion(mult = c(0.02, 0.12))
  ) +
  annotate(
    "label", x = 7, y = 43000,
    label = "Peak Infiltration\n(Day 7: +323% vs Sham, p < 0.001)",
    size = 3.2, fontface = "bold", color = "#CD534C",
    fill = "#FFF8F8"
  ) +
  theme(
    legend.position = "top",
    legend.justification = "left",
    panel.grid.major.y = element_line(color = "grey92", linewidth = 0.4, linetype = "dotted"),
    axis.text.x = element_text(size = 9.5, face = "bold", lineheight = 0.9)
  )
p_tc_mean

Alternatively, show_ribbon = TRUE provides continuous semi-transparent confidence envelopes around each treatment path:

Code 
# Smooth shaded mean ± SEM ribbon trajectories
p_tc_ribbon <- cliplot(
  expr ~ Day | Treatment,
  data = mpeg1,
  type = type_lines(
    linewidth        = 1.3,
    show_ribbon      = TRUE,
    ribbon_alpha     = 0.18,
    add_points       = TRUE,
    point_size       = 1.8,
    point_alpha      = 0.25,
    show_mean_points = TRUE,
    mean_point_size  = 3.4
  ),
  palette = "jco",
  title   = "Mpeg1 Trajectory with Continuous SEM Confidence Bands",
  subtitle = "Shaded Confidence Bands Displaying Between-Animal Dispersion",
  xlab    = "Days Post-Onset",
  ylab    = "Protein Expression (LFQ Intensity)"
) +
  scale_x_continuous(breaks = c(1, 3, 7, 14), labels = c("Day 1", "Day 3", "Day 7", "Day 14")) +
  scale_y_continuous(labels = label_number(scale = 1e-3, suffix = "k"), limits = c(0, NA)) +
  theme(
    legend.position = c(0.25, 0.82),
    legend.background = element_rect(fill = alpha("white", 0.85), color = "grey85", linewidth = 0.3)
  )
p_tc_ribbon
